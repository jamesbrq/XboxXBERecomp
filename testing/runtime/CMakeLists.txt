cmake_minimum_required(VERSION 3.25)
project(runtime_regressions LANGUAGES CXX)
enable_testing()
foreach(name stream_stop shader_cache door_routing)
    add_executable(${name} ${name}_test.cpp)
    target_include_directories(${name} PRIVATE ../../runtime)
    target_compile_features(${name} PRIVATE cxx_std_20)
    target_compile_options(${name} PRIVATE -UNDEBUG)
    add_test(NAME ${name} COMMAND ${name})
endforeach()
if(WIN32)
    add_executable(warm_shaders warm_shaders.cpp)
    target_include_directories(warm_shaders PRIVATE ../../runtime)
    target_compile_features(warm_shaders PRIVATE cxx_std_20)
    target_link_libraries(warm_shaders PRIVATE d3dcompiler)
endif()
